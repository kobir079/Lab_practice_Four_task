package org.example;

import java.sql.*;
import java.util.Scanner;

public class PrimeJDBC {

    static final String URL  = "jdbc:mysql://localhost:3306/primedb";
    static final String USER = "root";
    static final String PASS = "kobir@241184";

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter n: ");
        int n = sc.nextInt();

        if (n < 1) {
            System.out.println("n must be >= 1");
            return;
        }

        int count = 0, num = 1;
        while (count < n) {
            num++;
            boolean prime = true;
            for (int i = 2; i * i <= num; i++) {
                if (num % i == 0) { prime = false; break; }
            }
            if (prime) count++;
        }

        // database e save
        try (Connection con = DriverManager.getConnection(URL, USER, PASS)) {

            // table na thakle banabe
            try (Statement st = con.createStatement()) {
                st.executeUpdate("CREATE TABLE IF NOT EXISTS prime_store ("
                        + "id INT AUTO_INCREMENT PRIMARY KEY, "
                        + "n INT NOT NULL, "
                        + "nth_prime INT NOT NULL)");
            }

            try (PreparedStatement ps = con.prepareStatement(
                    "INSERT INTO prime_store (n, nth_prime) VALUES (?, ?)")) {
                ps.setInt(1, n);
                ps.setInt(2, num);
                ps.executeUpdate();
                System.out.println(n + " th prime = " + num + " -> saved!");
            }

        } catch (SQLException e) {
            System.out.println("Database error: " + e.getMessage());
        }
    }
}
